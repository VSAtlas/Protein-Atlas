# FUNCTION: run_last_stage
# Purpose:
#   Final-stage docking + consensus with:
#     Vina docking, GNINA docking, LeDock docking, GNINA CNN-rescore on Vina poses(this should actually be with the other rescoring functions)
#     pose validation on all engines, SCORCH rescoring, optional MM-GBSA, optional FEP/QM/MM,
#     rank-level consensus (RRF) with optional pose-agreement bonus.
#
# Inputs:
#   cfg: Dict
#   pdb_id: str
#   receptor_pdbqt: str
#   center: (float, float, float)
#   box_size: (float, float, float)
#   docking_mode: str  # "polypharmacology" | "discovery"
#   ligands: List[str] # paths to ligands entering this stage
#   logger: logging.Logger
#   score_history: Dict[str, Dict[str, Dict]]  # mutated in-place
#
# Outputs:
#   final_ranked_ligands: List[str]
#   final_consensus_scores: Dict[str, float]
#   score_history: Dict[...]  # updated

procedure run_last_stage(cfg, pdb_id, receptor_pdbqt, center, box_size, docking_mode, ligands, logger, score_history):

    # 0) Resolve final stage name from mode
    if docking_mode or cfg["MODE"] == "polypharmacology":
        last_stage_name := cfg.get("FINAL_STAGE_POLY", "stage3")
    else:
        last_stage_name := cfg.get("FINAL_STAGE_DISC", "stage5")

    if ligands is empty:
        logger.warning("[last-stage] No ligands; skipping.")
        return [], {}, score_history

    # Containers
    vina_scores := {};              vina_pose_paths := {};              vina_invalids := {}
    gnina_dock_scores := {};        gnina_dock_pose_paths := {};        gnina_dock_invalids := {}
    gnina_rescore_probs := {}
    ledock_scores := {};            ledock_pose_paths := {};            ledock_invalids := {}
    scorch_probs := {}

    # ---------------------------
    # 1) Vina DOCKING (pose gen)
    # ---------------------------
    if cfg.get("VINA_ENABLE", True):
        # *needs rewrite: run_one_stage
        # Change needed:
        #   - Accept params: engine_tag, stage_name, exhaustiveness, num_modes, vina_scoring
        #   - Write outputs under docked/<PDB_ID>/<stage_name>/vina/
        #   - Return (scores:Dict[lig->float], validated_pose_paths:Dict[lig->str], invalids:Dict[lig->str])
        # If not rewriting run_one_stage, call a new wrapper run_vina_stage (below).
        vina_scores, vina_pose_paths, vina_invalids :=
            run_vina_stage(
                cfg=cfg, pdb_id=pdb_id, receptor_pdbqt=receptor_pdbqt,
                center=center, box_size=box_size, ligands=ligands,
                last_stage_name=last_stage_name,
                exhaustiveness=cfg.get("FINAL_EXHAUSTIVENESS", 12),
                num_modes=cfg.get("FINAL_NUM_MODES", 5),
                vina_scoring=cfg.get("VINA_SCORING", "vina"),
                logger=logger
            )
        for lig in ligands:
            record_score(score_history, last_stage_name + "_vina", lig, vina_scores.get(lig), lig in vina_pose_paths, vina_invalids.get(lig, ""))
            record_le(score_history, last_stage_name + "_vina", lig, vina_scores.get(lig), heavy_atom_counts={})

    # -----------------------------------------------
    # 2) GNINA CNN RESCORE on Vina poses (no new dock)
    # -----------------------------------------------
    if cfg.get("GNINA_RESCORE_ENABLE", True) and vina_pose_paths is not empty:
        gnina_rescore_probs :=
            run_gnina_rescore_on_vina_poses(
                cfg=cfg, pdb_id=pdb_id, receptor_pdbqt=receptor_pdbqt,
                vina_pose_paths=vina_pose_paths,
                model_name=cfg.get("GNINA_MODEL_RESCORE", "crossdock_default2018"),
                out_dir_tag=last_stage_name + "/gnina_rescore",
                logger=logger
            )
        for lig, p in gnina_rescore_probs.items():
            record_score(score_history, last_stage_name + "_gnina_rescore", lig, p, True, "prob")

    # ------------------------------
    # 3) GNINA DOCKING (new poses)
    # ------------------------------
    if cfg.get("GNINA_DOCK_ENABLE", True):
        gnina_dock_scores, gnina_dock_pose_paths, gnina_dock_invalids :=
            run_gnina_dock_stage(
                cfg=cfg, pdb_id=pdb_id, receptor_pdbqt=receptor_pdbqt,
                center=center, box_size=box_size, ligands=ligands,
                last_stage_name=last_stage_name,
                model_name=cfg.get("GNINA_MODEL_DOCK", "crossdock_default2018"),
                exhaustiveness=cfg.get("GNINA_EXHAUSTIVENESS", 8),
                num_modes=cfg.get("GNINA_NUM_MODES", 5),
                logger=logger
            )
        for lig in ligands:
            record_score(score_history, last_stage_name + "_gnina_dock", lig, gnina_dock_scores.get(lig), lig in gnina_dock_pose_paths, gnina_dock_invalids.get(lig, ""))
            record_le(score_history, last_stage_name + "_gnina_dock", lig, gnina_dock_scores.get(lig), heavy_atom_counts={})

    # -------------------------
    # 4) LeDock DOCKING (poses)
    # -------------------------
    if cfg.get("LEDOCK_ENABLE", True):
        ledock_scores, ledock_pose_paths, ledock_invalids :=
            run_ledock_stage(
                cfg=cfg, pdb_id=pdb_id, receptor_pdbqt=receptor_pdbqt,
                center=center, box_size=box_size, ligands=ligands,
                last_stage_name=last_stage_name,
                logger=logger
            )
        for lig in ligands:
            record_score(score_history, last_stage_name + "_ledock", lig, ledock_scores.get(lig), lig in ledock_pose_paths, ledock_invalids.get(lig, ""))
            record_le(score_history, last_stage_name + "_ledock", lig, ledock_scores.get(lig), heavy_atom_counts={})

    # ----------------------------------------------------
    # 5) SCORCH rescoring on AVAILABLE VALIDATED POSES
    # ----------------------------------------------------
    if cfg.get("SCORCH_ENABLE", True):
        poses_for_scorch := {}
        for lig in ligands:
            if vina_pose_paths.get(lig):                poses_for_scorch[lig] := vina_pose_paths[lig]
            else if gnina_dock_pose_paths.get(lig):     poses_for_scorch[lig] := gnina_dock_pose_paths[lig]
            else if ledock_pose_paths.get(lig):         poses_for_scorch[lig] := ledock_pose_paths[lig]
        if poses_for_scorch not empty:
            scorch_probs :=
                run_scorch_rescore(
                    cfg=cfg, pdb_id=pdb_id, receptor_pdbqt=receptor_pdbqt,
                    pose_map=poses_for_scorch,
                    out_dir_tag=last_stage_name + "/scorch",
                    logger=logger
                )
            for lig, pr in scorch_probs.items():
                record_score(score_history, last_stage_name + "_scorch", lig, pr, True, "prob")

    # -----------------------------------------------------------------
    # 6) Build engine ranks (+ optional pose-agreement bonus) ? RRF
    # -----------------------------------------------------------------
    dict_of_scores := {};  better_flags := {}
    if vina_scores:            dict_of_scores["vina"] := vina_scores;               better_flags["vina"] := False
    if gnina_rescore_probs:    dict_of_scores["gnina_rescore"] := gnina_rescore_probs; better_flags["gnina_rescore"] := True
    if gnina_dock_scores:      dict_of_scores["gnina_dock"] := gnina_dock_scores;   better_flags["gnina_dock"] := True
    if ledock_scores:          dict_of_scores["ledock"] := ledock_scores;           better_flags["ledock"] := False
    if scorch_probs:           dict_of_scores["scorch"] := scorch_probs;            better_flags["scorch"] := True

    engine_ranks :=
        build_engine_ranks(dict_of_scores, better_flags)

    engine_ranks :=
        apply_pose_agreement_bonus(
            engine_ranks=engine_ranks,
            pose_paths_by_engine={
                "vina": vina_pose_paths,
                "gnina_dock": gnina_dock_pose_paths,
                "ledock": ledock_pose_paths
            },
            rmsd_cutoff=cfg.get("POSE_AGREEMENT_RMSD", 2.0),
            logger=logger
        )

    fused := compute_rrf_consensus(engine_ranks, k=cfg.get("RRF_K", 60))
    final_ranked_ligands := sort ligands by fused[lig] descending
    final_consensus_scores := fused
    for lig in final_ranked_ligands:
        record_score(score_history, last_stage_name + "_consensus_rrf", lig, final_consensus_scores[lig], True, "fused")

    # ---------------------------------------
    # 7) Optional physics: MM-GBSA ? FEP/QM/MM
    # ---------------------------------------
    if cfg.get("MM_GBSA_ENABLE", False):
        top_pct := cfg.get("MM_GBSA_TOP_PCT", 0.10)
        n_top := max(1, int(len(final_ranked_ligands) * top_pct))
        mmgbsa_candidates := first n_top of final_ranked_ligands

        mmgbsa_pose_map := {}
        for lig in mmgbsa_candidates:
            if vina_pose_paths.get(lig):                mmgbsa_pose_map[lig] := vina_pose_paths[lig]
            else if gnina_dock_pose_paths.get(lig):     mmgbsa_pose_map[lig] := gnina_dock_pose_paths[lig]
            else if ledock_pose_paths.get(lig):         mmgbsa_pose_map[lig] := ledock_pose_paths[lig]

        mmgbsa_scores :=
            run_mmgbsa_rescore(
                cfg=cfg, pdb_id=pdb_id, receptor_pdbqt=receptor_pdbqt,
                pose_map=mmgbsa_pose_map,
                out_dir_tag=last_stage_name + "/physics/mmgbsa",
                logger=logger
            )
        for lig, dG in mmgbsa_scores.items():
            record_score(score_history, last_stage_name + "_mmgbsa", lig, dG, True, "deltaG")

        if cfg.get("FEP_QMMM_ENABLE", False):
            y_pct := cfg.get("FEP_QMMM_TOP_PCT", 0.20)
            mmgbsa_sorted := ligands in mmgbsa_scores sorted by ascending ?G
            n_y := max(1, int(len(mmgbsa_sorted) * y_pct))
            fep_set := first n_y of mmgbsa_sorted

            fep_pose_map := {}
            for lig in fep_set:
                if vina_pose_paths.get(lig):            fep_pose_map[lig] := vina_pose_paths[lig]
                else if gnina_dock_pose_paths.get(lig): fep_pose_map[lig] := gnina_dock_pose_paths[lig]
                else if ledock_pose_paths.get(lig):     fep_pose_map[lig] := ledock_pose_paths[lig]

            final_physics_scores :=
                run_fep_qmmm_final(
                    cfg=cfg, pdb_id=pdb_id, receptor_pdbqt=receptor_pdbqt,
                    pose_map=fep_pose_map,
                    out_dir_tag=last_stage_name + "/physics/fep_qmmm",
                    logger=logger
                )
            for lig, dGf in final_physics_scores.items():
                record_score(score_history, last_stage_name + "_fep_qmmm", lig, dGf, True, "deltaG_final")

            if cfg.get("FINAL_SORT_BY_PHYSICS", True):
                final_ranked_ligands :=
                    reorder_by_physics_then_fused(final_ranked_ligands, final_physics_scores, fused)
                for idx, lig in enumerate(final_ranked_ligands, start=1):
                    record_score(score_history, last_stage_name + "_final", lig, float(idx), True, "rank_final")

    return final_ranked_ligands, final_consensus_scores, score_history
