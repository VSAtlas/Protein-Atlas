# Directory containing your logs
$root = 'E:\10242025'

# Patterns to drop
$patterns = @(
  '^\s*Docking \(stage\d+\):\s*\d+%',      # tqdm/progress lines
  '^\[?vina-call\]',                        # vina-call lines
  'INFO\s+-\s+.*?:\s+RMSD filter kept\b',   # RMSD filter lines
  'INFO\s+-\s+.*?\.pdbqt validation:',      # validation dict lines
  '^\[\d{2}:\d{2}:\d{2}\]',                 # RDKit timestamps like [21:43:43]
  '^\*\*\*\s+Open Babel Warning',           # Open Babel warnings
  '^\s*Failed to kekulize',                 # kekulization note
  '^\s*ERROR:',                             # ERROR: ...
  '^\s*1 molecule converted\b',             # OBabel summary
  '^\s*=+\s*$'                              # ===== separators
)
$escAnsi = '\x1B\[[0-9;]*[A-Za-z]'          # strip stray ANSI sequences

Get-ChildItem -LiteralPath $root -Filter 'MAIN_*.log' -File | ForEach-Object {
  $in  = $_.FullName
  $out = Join-Path $_.DirectoryName ($_.BaseName + '.cleaned' + $_.Extension)

  $script:blank = 0  # reset per file
  Get-Content -LiteralPath $in | Where-Object {
    $line = $_
    foreach ($pat in $patterns) { if ($line -match $pat) { return $false } }
    $true
  } | ForEach-Object { $_ -replace $escAnsi, '' } | Where-Object {
    if ($_ -match '^\s*$') { $script:blank++; return ($script:blank -le 1) } else { $script:blank = 0; return $true }
  } | Set-Content -LiteralPath $out -Encoding UTF8

  Write-Host "Cleaned log -> $out"
}
