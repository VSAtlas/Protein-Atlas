Commit iterating  0.01 at a time
base="brcf-update-FAST_EDITION-v"
i=1  # 1 == 0.01
while true; do
  vers=$(printf "%d.%02d" $((i/100)) $((i%100)))   # 1 -> 0.01, 2 -> 0.02, ..., 100 -> 1.00
  candidate="${base}${vers}"
  git show-ref --verify --quiet "refs/heads/${candidate}" || \
  git ls-remote --exit-code --heads origin "${candidate}" >/dev/null 2>&1 || break
  i=$((i+1))
done
branch="${candidate}"
git switch -c "$branch" 2>/dev/null || git switch "$branch"
( git diff --quiet && git diff --cached --quiet ) || ( git add -A && git commit -m "$branch" )
git push -u origin "$branch"
